D1 RESOURCES

Code & formulas I reach for often

Snippets I reuse across projects — copy and adapt them for your own workbooks.

calendar-table.pq Power Query M 
let
    StartDate = #date(2020, 1, 1),
    EndDate = #date(2030, 12, 31),
    DayCount = Duration.Days(EndDate - StartDate) + 1,
    DateList = List.Dates(StartDate, DayCount, #duration(1,0,0,0)),
    #"Converted to Table" = Table.FromList(DateList, Splitter.SplitByNothing(), {"Date"}),
    #"Added Year" = Table.AddColumn(#"Converted to Table", "Year", each Date.Year([Date])),
    #"Added Month" = Table.AddColumn(#"Added Year", "Month", each Date.Month([Date])),
    #"Added MonthName" = Table.AddColumn(#"Added Month", "MonthName", each Date.ToText([Date], "MMMM")),
    #"Added Quarter" = Table.AddColumn(#"Added MonthName", "Quarter", each "Q" & Text.From(Date.QuarterOfYear([Date]))),
    #"Added WeekdayName" = Table.AddColumn(#"Added Quarter", "WeekdayName", each Date.ToText([Date], "dddd")),
    #"Added IsWeekend" = Table.AddColumn(#"Added WeekdayName", "IsWeekend", each Date.DayOfWeek([Date], Day.Monday) >= 5)
in
    #"Added IsWeekend"
dynamic-lookup.fx Excel Formula 
// XLOOKUP with a graceful fallback instead of #N/A
=IFERROR(XLOOKUP([@SKU], Prices[SKU], Prices[UnitPrice]), "Not found")
rolling-12-month.fx Excel Formula 
// Rolling 12-month total ending on the row's date
=SUMIFS(Sales[Amount], Sales[Date], ">="&EDATE([@Date],-11), Sales[Date], "<="&[@Date])
